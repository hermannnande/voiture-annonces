# ========================================
# VARIABLES GOOGLE OAUTH À AJOUTER DANS RAILWAY
# ========================================

# 1. Créez d'abord votre projet OAuth sur Google Cloud Console
#    https://console.cloud.google.com/

# 2. Copiez le Client ID et Client Secret que Google vous donne

# 3. Ajoutez ces 3 variables dans Railway :

GOOGLE_CLIENT_ID=VOTRE_CLIENT_ID_ICI
GOOGLE_CLIENT_SECRET=VOTRE_CLIENT_SECRET_ICI
GOOGLE_CALLBACK_URL=https://voiture-annonces-production.up.railway.app/api/auth/google/callback

# 4. Vérifiez aussi que cette variable existe (pour la redirection après login) :
FRONTEND_URL=https://voiture-annonces.vercel.app

# ========================================
# URLS À CONFIGURER DANS GOOGLE CLOUD CONSOLE
# ========================================

# Dans "Authorized JavaScript origins" :
https://voiture-annonces.vercel.app
https://voiture-annonces-production.up.railway.app
http://localhost:3000

# Dans "Authorized redirect URIs" :
https://voiture-annonces-production.up.railway.app/api/auth/google/callback
http://localhost:3001/api/auth/google/callback

# ========================================
# POUR TESTER EN LOCAL (OPTIONNEL)
# ========================================

# Si vous voulez tester en local, créez un fichier backend/.env avec :
GOOGLE_CLIENT_ID=votre_client_id
GOOGLE_CLIENT_SECRET=votre_client_secret
GOOGLE_CALLBACK_URL=http://localhost:3001/api/auth/google/callback
FRONTEND_URL=http://localhost:3000
DATABASE_URL=votre_database_url
JWT_SECRET=votre_jwt_secret
JWT_REFRESH_SECRET=votre_refresh_secret
JWT_EXPIRATION=24h
JWT_REFRESH_EXPIRATION=7d
NODE_ENV=development
PORT=3001
